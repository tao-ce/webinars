\subsection{Look \& Feel}
\begin{frame}[fragile]
    \begin{description}
    \item[Goal] Change logo in \alert{Portal} application.
    \item[How] \begin{enumerate}
        \item Add your logo in \texttt{/opt/tao-ce/portal/assets/images} directory
        \item Edit \texttt{portal.configuration} in \href{https://github.com/tao-ce/tao-ce/blob/837b29614f205a40918327fa478b5c0170e5b6d1/libexec/setup/apps/files/environments.libsonnet#L19-L21}{\texttt{environments.libsonnet}}
        \item Change \texttt{logoImage}
    \end{enumerate}
    \item[Rebuild] \,\begin{lstlisting}
# Apply change in config templates, and regenerate config files
task dev:setup
# Restart EM services
systemctl restart tao-ce.env\*
\end{lstlisting}
    \end{description}

\subsection{Benchmark \taoce}
\begin{frame}[fragile]
    \begin{description}
    \item[Goal] Ensure \taoce can support 100 concurrent test-takers
    \item[How] \, \begin{enumerate}
        \item prepare benchmark users
        \item assign a session
        \item run benchmark in \href{https://github.com/tao-ce/tao-ce/tree/main/tests/load}{\texttt{tests/load} directory}
    \end{enumerate}
    \item[Run] \,\begin{lstlisting}
. ./.venv/bin/activate
locust \
    -H https://community.tao.internal \
    --load_profile 'Light (100u)' \
    --wait_time 'Fast (0.5-2s)' \
    --ramp_method normal --ramp_duration 30
\end{lstlisting}
    \end{description}
\end{frame}

\subsection{Do you even need to rebuild?}
\begin{frame}[fragile]
    \only<article>{%
    \taoce being shipped as container images, you can take advantage of existing images.
    }
    \begin{description}
    \item[Goal] Apply minor changes to \taoce configuration, without complete rebuild.
    \item[How] \, \begin{enumerate}
        \item create a new \texttt{my.taoce.Containerfile} file \begin{lstlisting}
FROM quay.io/tao-ce/tao-ce:lite

COPY path/to/my/changes \
    /usr/local/libexec/tao-ce/...
RUN my/custom/script.sh ...
\end{lstlisting}
        \item use it in \href{https://github.com/tao-ce/tao-ce/blob/837b29614f205a40918327fa478b5c0170e5b6d1/docker-compose.yml#L9}{\texttt{docker-compose.yml}}  \begin{lstlisting}
services:
  tao:
    build:     # replace image by build
      context: .
      dockerfile: my.taoce.Containerfile
\end{lstlisting}
    \end{enumerate}
    \item[Reinstall] \,\begin{lstlisting}
docker compose down -v
docker compose up -d
\end{lstlisting}
    \end{description}
\end{frame}
