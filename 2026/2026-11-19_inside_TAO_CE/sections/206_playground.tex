        \item Edit \texttt{portal.configuration} in \href{https://github.com/tao-ce/tao-ce/blob/837b29614f205a40918327fa478b5c0170e5b6d1/libexec/setup/apps/files/environments.libsonnet#L19-L21}{\texttt{environments.libsonnet}}

\subsection{Do you even need to rebuild?}
\begin{frame}[fragile=singleslide]
    \only<article>{\taoce being shipped as container images, you can take advantage of existing images.}
    \begin{description}
    \item[Goal] Apply minor changes to \taoce configuration, without complete rebuild.
    \item[How] \, \begin{enumerate}
        \item create a new \texttt{my.taoce.Containerfile} file \inputminted{Dockerfile}{./snippets/my.taoce.Containerfile}
        \item use it in \href{https://github.com/tao-ce/tao-ce/blob/837b29614f205a40918327fa478b5c0170e5b6d1/docker-compose.yml#L9}{\texttt{docker-compose.yml}}
    \inputminted{yaml}{./snippets/dc.yaml}
    \end{enumerate}
    \item[Reinstall] \mintinline{bash}{docker compose down -v && docker compose up -d}
    \end{description}
\end{frame}
